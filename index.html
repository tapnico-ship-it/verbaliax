function parseResult(raw) { const isEs = state.lang === 'es'; const marker = isEs ? 'TEXTO_CORREGIDO:' : 'CORRECTED_TEXT:'; const expMarker = isEs ? 'EXPLICACIONES:' : 'EXPLANATIONS:'; let corrected = ""; let explanations = []; if (raw.includes(marker)) { const parts = raw.split(marker)[1].split(expMarker); corrected = parts[0].trim(); if (parts[1]) { explanations = parts[1].trim().split('\n') .map(line => line.replace(/^[\s\d.-]+/, '').trim()) // Limpia números, guiones y espacios al inicio .filter(line => line.length > 3); } } else { corrected = raw.trim(); } return { corrected: corrected || raw, explanations: explanations.slice(0, 5) }; } async function correctText() { const text = el.inputText.value.trim(); if (!text) return; if (!state.apiKey) { alert(i18n[state.lang].apiKeyPrompt); el.apiModal.classList.remove('hidden'); return; } el.mainLoader.style.display = 'inline-block'; el.btnCorrect.disabled = true; try { const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${state.apiKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: buildPrompt(text) }] }] }) }); const data = await response.json(); if (data.error) throw new Error(data.error.message); const rawResponse = data.candidates[0].content.parts[0].text; const result = parseResult(rawResponse); displayResult(result); } catch (error) { console.error(error); alert(i18n[state.lang].apiError); } finally { el.mainLoader.style.display = 'none'; el.btnCorrect.disabled = false; } } function displayResult(result) { el.outputText.textContent = result.corrected; updateCounts(result.corrected, el.outputCounter); if (state.tutor) { if (result.explanations.length > 0) { el.tutorContent.innerHTML = result.explanations.map((exp, i) => `
${i + 1}. ${escapeHtml(exp)}
`).join(''); } else { el.tutorContent.innerHTML = `

${i18n[state.lang].noErrors}

`; } } } function updateCounts(text, displayEl) { const words = text.trim() ? text.trim().split(/\s+/).length : 0; displayEl.textContent = `${words} ${state.lang === 'es' ? 'palabras' : 'words'}`; } function escapeHtml(text) { const div = document.createElement('div'); div.textContent = text; return div.innerHTML; } // Event Listeners el.langSelect.addEventListener('change', (e) => { state.lang = e.target.value; localStorage.setItem('verbaliax_lang', state.lang); updateUI(); }); el.tutorToggle.addEventListener('change', (e) => { state.tutor = e.target.checked; localStorage.setItem('verbaliax_tutor', state.tutor); el.tutorPanel.classList.toggle('hidden', !state.tutor); }); el.inputText.addEventListener('input', () => { updateCounts(el.inputText.value, el.inputCounter); }); el.btnClear.addEventListener('click', () => { el.inputText.value = ''; el.outputText.textContent = ''; el.tutorContent.innerHTML = `

${i18n[state.lang].tutorPlaceholder}

`; updateCounts('', el.inputCounter); updateCounts('', el.outputCounter); }); el.btnCorrect.addEventListener('click', correctText); el.btnCopy.addEventListener('click', () => { const text = el.outputText.textContent; if (!text) return; navigator.clipboard.writeText(text).then(() => { const originalText = el.txtCopy.textContent; el.txtCopy.textContent = i18n[state.lang].copySuccess; setTimeout(() => el.txtCopy.textContent = originalText, 2000); }); }); el.btnDownload.addEventListener('click', () => { const text = el.outputText.textContent; if (!text) return; const blob = new Blob([text], { type: 'text/plain' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = `verbaliax_corrected_${new Date().getTime()}.txt`; a.click(); }); el.settingsBtn.addEventListener('click', () => { el.apiKeyInput.value = state.apiKey; el.apiModal.classList.remove('hidden'); }); el.apiSave.addEventListener('click', () => { state.apiKey = el.apiKeyInput.value.trim(); localStorage.setItem('verbaliax_api_key', state.apiKey); el.apiModal.classList.add('hidden'); }); el.apiClose.addEventListener('click', () => el.apiModal.classList.add('hidden')); // Init updateUI(); if (!state.apiKey) setTimeout(() => el.apiModal.classList.remove('hidden'), 1000);
